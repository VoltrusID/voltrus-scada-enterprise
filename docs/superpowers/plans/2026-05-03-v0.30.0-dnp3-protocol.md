// src/dnp3/datalink.rs

use anyhow::{Context, Result};

// CRC-CCITT lookup table (polynomial 0x1021)
const CRC_TABLE: [u16; 256] = {
    let mut table = [0u16; 256];
    let mut i = 0u16;
    while i < 256 {
        let mut crc = i << 8;
        let mut j = 0;
        while j < 8 {
            if crc & 0x8000 != 0 {
                crc = (crc << 1) ^ 0x1021;
            } else {
                crc <<= 1;
            }
            j += 1;
        }
        table[i as usize] = crc;
        i += 1;
    }
    table
};

pub fn crc_ccitt(data: &[u8]) -> u16 {
    let mut crc: u16 = 0xFFFF;
    for &byte in data {
        crc = (crc << 8) ^ CRC_TABLE[((crc >> 8) ^ byte as u16) as usize];
    }
    crc
}

/// DNP3 data link control byte.
#[derive(Debug, Clone, PartialEq)]
pub struct ControlByte {
    pub master: bool,   // DIR: 1 = from master
    pub fir: bool,      // FCB: frame count bit
    pub fin: bool,      // FCV: frame count valid
    pub sequence: u8,   // 3-bit sequence
}

impl ControlByte {
    pub fn to_byte(&self) -> u8 {
        let dir = if self.master { 0x80 } else { 0x00 };
        let prm = if self.master { 0x40 } else { 0x00 };
        let fcb = if self.fir { 0x20 } else { 0x00 };
        let fcv = if self.fin { 0x10 } else { 0x00 };
        let func = self.sequence & 0x0F;
        dir | prm | fcb | fcv | func
    }

    pub fn from_byte(byte: u8) -> Self {
        Self {
            master: byte & 0x80 != 0,
            fir: byte & 0x20 != 0,
            fin: byte & 0x10 != 0,
            sequence: byte & 0x0F,
        }
    }
}

/// DNP3 data link layer frame.
#[derive(Debug, Clone, PartialEq)]
pub struct Dnp3Frame {
    pub master: bool,
    pub fir: bool,
    pub fin: bool,
    pub sequence: u8,
    pub destination: u16,
    pub source: u16,
    pub data: Vec<u8>,
}

impl Dnp3Frame {
    /// Encode frame to bytes with CRC blocks.
    /// DNP3 inserts CRC-CCITT after every 16 bytes of user data.
    pub fn encode(&self) -> Vec<u8> {
        let ctrl = ControlByte {
            master: self.master,
            fir: self.fir,
            fin: self.fin,
            sequence: self.sequence,
        };
        let user_data_len = 5 + self.data.len(); // ctrl(1) + dest(2) + src(2) + payload
        let total_len = user_data_len + 2; // +2 for start bytes + length byte not in user data

        let mut raw = Vec::with_capacity(total_len + 10);
        // Start bytes
        raw.push(0x05);
        raw.push(0x64);
        // Length: user data bytes (ctrl + dest + src + data)
        raw.push(user_data_len as u8);
        // Control byte
        raw.push(ctrl.to_byte());
        // Destination address (little-endian)
        raw.push((self.destination & 0xFF) as u8);
        raw.push(((self.destination >> 8) & 0xFF) as u8);
        // Source address (little-endian)
        raw.push((self.source & 0xFF) as u8);
        raw.push(((self.source >> 8) & 0xFF) as u8);
        // Payload
        raw.extend_from_slice(&self.data);

        // Insert CRC blocks every 16 bytes (excluding start bytes and length)
        insert_crc_blocks(&mut raw)
    }

    /// Decode bytes (with CRC blocks) into a frame.
    pub fn decode(bytes: &[u8]) -> Result<Self> {
        if bytes.len() < 10 {
            anyhow::bail!("Frame too short: {} bytes", bytes.len());
        }
        // Verify start bytes
        if bytes[0] != 0x05 || bytes[1] != 0x64 {
            anyhow::bail!("Invalid start bytes: {:02X} {:02X}", bytes[0], bytes[1]);
        }

        // Strip CRC blocks to get raw data
        let raw = strip_crc_blocks(bytes)?;

        let len = raw[2] as usize;
        if raw.len() < 3 + len {
            anyhow::bail!("Frame truncated: expected {} bytes, got {}", len, raw.len() - 3);
        }

        let ctrl = ControlByte::from_byte(raw[3]);
        let destination = raw[4] as u16 | ((raw[5] as u16) << 8);
        let source = raw[6] as u16 | ((raw[7] as u16) << 8);
        let data = raw[8..3 + len].to_vec();

        Ok(Self {
            master: ctrl.master,
            fir: ctrl.fir,
            fin: ctrl.fin,
            sequence: ctrl.sequence,
            destination,
            source,
            data,
        })
    }
}

/// Insert CRC-CCITT after every 16 bytes of user data (after start+length).
fn insert_crc_blocks(raw: &mut Vec<u8>) -> Vec<u8> {
    let mut result = Vec::new();
    // Start bytes + length are always first 3 bytes, no CRC
    result.extend_from_slice(&raw[0..3]);

    let user_data = &raw[3..];
    let mut offset = 0;
    while offset < user_data.len() {
        let block_end = std::cmp::min(offset + 16, user_data.len());
        let block = &user_data[offset..block_end];
        result.extend_from_slice(block);
        let crc = crc_ccitt(block);
        result.push((crc & 0xFF) as u8);
        result.push(((crc >> 8) & 0xFF) as u8);
        offset = block_end;
    }

    result
}

/// Strip CRC blocks from a received frame, returning raw bytes (start + length + data).
fn strip_crc_blocks(bytes: &[u8]) -> Result<Vec<u8>> {
    if bytes.len() < 6 {
        anyhow::bail!("Frame too short for CRC stripping");
    }

    let mut result = Vec::new();
    // First 3 bytes are start(2) + length(1) — no CRC
    result.extend_from_slice(&bytes[0..3]);

    // Remaining: groups of (up to 16 data bytes + 2 CRC bytes)
    let mut offset = 3;
    while offset < bytes.len() {
        // Read up to 16 data bytes + 2 CRC bytes
        let remaining = bytes.len() - offset;
        if remaining <= 2 {
            break;
        }
        let data_len = std::cmp::min(remaining - 2, 16);
        let data = &bytes[offset..offset + data_len];
        let crc_bytes = &bytes[offset + data_len..offset + data_len + 2];
        let expected_crc = crc_ccitt(data);
        let actual_crc = crc_bytes[0] as u16 | ((crc_bytes[1] as u16) << 8);
        if expected_crc != actual_crc {
            anyhow::bail!(
                "CRC mismatch at offset {}: expected {:04X}, got {:04X}",
                offset,
                expected_crc,
                actual_crc
            );
        }
        result.extend_from_slice(data);
        offset += data_len + 2;
    }

    Ok(result)
}

/// Verify CRC blocks in a byte stream.
fn verify_crc_blocks(bytes: &[u8]) -> bool {
    strip_crc_blocks(bytes).is_ok()
}

use anyhow::Result;
use std::sync::Arc;

use crate::cache::DeviceCache;
use crate::models::Dnp3DeviceConfig;

pub async fn poll_dnp3_device(config: Dnp3DeviceConfig, cache: DeviceCache) {
    let device_id = config.id.clone();
    let mut backoff = Backoff::new();

    loop {
        match try_poll(&config, &cache).await {
            Ok(()) => backoff.reset(),
            Err(e) => {
                tracing::warn!("DNP3 device {device_id} error: {e}");
                cache.set_status(&device_id, crate::models::DeviceStatus::Offline).await;
                let delay = backoff.next();
                tracing::info!("DNP3 device {device_id} retrying in {delay}ms");
                tokio::time::sleep(std::time::Duration::from_millis(delay)).await;
            }
        }
    }
}

async fn try_poll(config: &Dnp3DeviceConfig, cache: &DeviceCache) -> Result<()> {
    let mut session = master::Dnp3MasterSession::connect(
        &config.ip,
        config.port,
        config.master_address,
        config.outstation_address,
    )
    .await?;

    tracing::info!("DNP3 device {} connected", config.id);
    cache.set_status(&config.id, crate::models::DeviceStatus::Online).await;

    // Initial integrity poll
    let values = session.integrity_poll().await?;
    let mut metrics = std::collections::HashMap::new();
    for point in &config.points {
        if let Some(val) = values.get(&(point.group_num, point.variation, point.point_index)) {
            metrics.insert(point.name.clone(), val.value * point.scale);
        }
    }
    cache.update(&config.id, metrics).await;

    // Periodic event scan loop
    let scan_interval = config.event_scan_interval_ms.unwrap_or(1000);
    loop {
        tokio::time::sleep(std::time::Duration::from_millis(scan_interval)).await;
        let events = session.event_scan(1).await?;
        let mut metrics = std::collections::HashMap::new();
        for point in &config.points {
            if let Some(val) = events.get(&(point.group_num, point.variation, point.point_index)) {
                metrics.insert(point.name.clone(), val.value * point.scale);
            }
        }
        if !metrics.is_empty() {
            cache.update(&config.id, metrics).await;
        }
    }
}

struct Backoff {
    attempt: u32,
    max_delay_ms: u64,
}

impl Backoff {
    fn new() -> Self {
        Self {
            attempt: 0,
            max_delay_ms: 30_000,
        }
    }

    fn reset(&mut self) {
        self.attempt = 0;
    }

    fn next(&mut self) -> u64 {
        let attempt = self.attempt.min(30);
        let delay = 1000 * 2u64.pow(attempt);
        self.attempt = self.attempt.saturating_add(1);
        delay.min(self.max_delay_ms)
    }
}

// src/dnp3/transport.rs

/// DNP3 transport header (1 byte: FIR | FIN | SEQUENCE).
#[derive(Debug, Clone, PartialEq)]
pub struct TransportHeader {
    pub fir: bool,
    pub fin: bool,
    pub sequence: u8,
}

impl TransportHeader {
    pub fn to_byte(&self) -> u8 {
        let fir = if self.fir { 0x80 } else { 0x00 };
        let fin = if self.fin { 0x40 } else { 0x00 };
        fir | fin | (self.sequence & 0x3F)
    }

    pub fn from_byte(byte: u8) -> Self {
        Self {
            fir: byte & 0x80 != 0,
            fin: byte & 0x40 != 0,
            sequence: byte & 0x3F,
        }
    }
}

/// A single transport segment extracted from a data link frame.
#[derive(Debug, Clone)]
pub struct TransportSegment {
    pub fir: bool,
    pub fin: bool,
    pub sequence: u8,
    pub data: Vec<u8>,
}

/// Reassembles multi-segment ASDUs from transport segments.
#[derive(Debug)]
pub struct TransportReassembler {
    buffer: Vec<u8>,
    expected_seq: u8,
    active: bool,
}

impl TransportReassembler {
    pub fn new() -> Self {
        Self {
            buffer: Vec::new(),
            expected_seq: 0,
            active: false,
        }
    }

    /// Process a transport segment. Returns complete ASDU data when FIN=true.
    pub fn process(&mut self, segment: TransportSegment) -> Option<Vec<u8>> {
        // New FIR resets any partial reassembly
        if segment.fir {
            self.buffer.clear();
            self.active = true;
            self.expected_seq = segment.sequence;
        }

        if !self.active {
            return None;
        }

        // Verify sequence
        if segment.sequence != self.expected_seq {
            tracing::warn!(
                "Transport sequence mismatch: expected {}, got {}",
                self.expected_seq,
                segment.sequence
            );
            self.buffer.clear();
            self.active = false;
            return None;
        }

        self.buffer.extend_from_slice(&segment.data);
        self.expected_seq = (self.expected_seq + 1) & 0x3F;

        if segment.fin {
            let result = self.buffer.clone();
            self.buffer.clear();
            self.active = false;
            return Some(result);
        }

        None
    }
}

/// Parse packed binary input (Group 1 Var 1).
/// Each byte contains 8 binary points, LSB = first point.
pub fn parse_binary_input_packed(data: &[u8], start_index: u16) -> Vec<Dnp3PointValue> {
    let mut values = Vec::new();
    for (byte_idx, &byte) in data.iter().enumerate() {
        for bit in 0..8u16 {
            let index = start_index + (byte_idx as u16) * 8 + bit;
            values.push(Dnp3PointValue {
                group: 1,
                variation: 1,
                index,
                value: if byte & (1 << bit) != 0 { 1.0 } else { 0.0 },
            });
        }
    }
    values
}

/// Parse binary input with flags (Group 1 Var 2).
/// Each point: 1 byte flags, optional time (not parsed here).
pub fn parse_binary_input_flags(data: &[u8], start_index: u16, count: u16) -> Vec<Dnp3PointValue> {
    let mut values = Vec::new();
    for i in 0..count as usize {
        let offset = i; // 1 byte per point (flags only, no timestamp)
        if offset < data.len() {
            let flags = data[offset];
            values.push(Dnp3PointValue {
                group: 1,
                variation: 2,
                index: start_index + i as u16,
                value: if flags & 0x01 != 0 { 1.0 } else { 0.0 },
            });
        }
    }
    values
}

/// Parse analog input int32 (Group 30 Var 1 or 3).
/// Var 1: 1 byte flags + 4 bytes int32. Var 3: 4 bytes int32 only.
pub fn parse_analog_input_int32(data: &[u8], start_index: u16, count: u16) -> Vec<Dnp3PointValue> {
    let mut values = Vec::new();
    for i in 0..count as usize {
        let offset = i * 4;
        if offset + 4 <= data.len() {
            let raw = u32::from_le_bytes([data[offset], data[offset + 1], data[offset + 2], data[offset + 3]]);
            values.push(Dnp3PointValue {
                group: 30,
                variation: 3,
                index: start_index + i as u16,
                value: raw as i32 as f64,
            });
        }
    }
    values
}

use anyhow::{Context, Result};

/// DNP3 application function codes.
#[derive(Debug, Clone, Copy, PartialEq)]
#[repr(u8)]
pub enum FunctionCode {
    Read = 0x01,
    Write = 0x02,
    Select = 0x03,
    Operate = 0x04,
    DirectOperate = 0x05,
    DirectOperateNoAck = 0x06,
    Freeze = 0x07,
    FreezeClear = 0x08,
    FreezeAtTime = 0x09,
    ColdRestart = 0x0D,
    WarmRestart = 0x0E,
    EnableUnsolicited = 0x14,
    DisableUnsolicited = 0x15,
    Response = 0x81,
    UnsolicitedResponse = 0x82,
}

impl FunctionCode {
    pub fn from_u8(v: u8) -> Result<Self> {
        match v {
            0x01 => Ok(Self::Read),
            0x02 => Ok(Self::Write),
            0x03 => Ok(Self::Select),
            0x04 => Ok(Self::Operate),
            0x05 => Ok(Self::DirectOperate),
            0x06 => Ok(Self::DirectOperateNoAck),
            0x07 => Ok(Self::Freeze),
            0x08 => Ok(Self::FreezeClear),
            0x09 => Ok(Self::FreezeAtTime),
            0x0D => Ok(Self::ColdRestart),
            0x0E => Ok(Self::WarmRestart),
            0x14 => Ok(Self::EnableUnsolicited),
            0x15 => Ok(Self::DisableUnsolicited),
            0x81 => Ok(Self::Response),
            0x82 => Ok(Self::UnsolicitedResponse),
            _ => anyhow::bail!("Unknown function code: 0x{:02X}", v),
        }
    }
}

/// DNP3 object header qualifier codes.
#[derive(Debug, Clone, Copy, PartialEq)]
#[repr(u8)]
pub enum QualifierCode {
    StartStop8 = 0x00,
    StartStop16 = 0x01,
    Count8 = 0x07,
    Count16 = 0x08,
    CountAndPrefix8 = 0x17,
}

/// DNP3 Internal Indications (IIN) — 2 bytes of device status flags.
#[derive(Debug, Clone, Default)]
pub struct Iin {
    pub all_stations: bool,
    pub device_restart: bool,
    pub device_trouble: bool,
    pub local_control: bool,
    pub need_time: bool,
    pub class_1_events: bool,
    pub class_2_events: bool,
    pub class_3_events: bool,
}

impl Iin {
    pub fn to_low_byte(&self) -> u8 {
        let mut byte = 0u8;
        if self.all_stations { byte |= 0x01; }
        if self.device_restart { byte |= 0x02; }
        if self.device_trouble { byte |= 0x04; }
        if self.local_control { byte |= 0x08; }
        if self.need_time { byte |= 0x10; }
        if self.class_1_events { byte |= 0x20; }
        if self.class_2_events { byte |= 0x40; }
        if self.class_3_events { byte |= 0x80; }
        byte
    }

    pub fn from_bytes(low: u8, _high: u8) -> Self {
        Self {
            all_stations: low & 0x01 != 0,
            device_restart: low & 0x02 != 0,
            device_trouble: low & 0x04 != 0,
            local_control: low & 0x08 != 0,
            need_time: low & 0x10 != 0,
            class_1_events: low & 0x20 != 0,
            class_2_events: low & 0x40 != 0,
            class_3_events: low & 0x80 != 0,
        }
    }
}

/// DNP3 ASDU header (4 bytes: function, control, IIN low, IIN high).
#[derive(Debug, Clone)]
pub struct AsduHeader {
    pub function: FunctionCode,
    pub sequence: u8,
    pub iin: Iin,
}

impl AsduHeader {
    pub fn encode(&self) -> [u8; 4] {
        let ac = 0xC0 | (self.sequence & 0x0F); // FIR=1, FIN=1, CON=0
        [
            self.function as u8,
            ac,
            self.iin.to_low_byte(),
            0x00, // IIN high byte (not used for requests)
        ]
    }

    pub fn decode(bytes: &[u8; 4]) -> Result<Self> {
        let function = FunctionCode::from_u8(bytes[0])?;
        let sequence = bytes[1] & 0x0F;
        let iin = Iin::from_bytes(bytes[2], bytes[3]);
        Ok(Self { function, sequence, iin })
    }
}

/// DNP3 object header in an ASDU.
#[derive(Debug, Clone)]
pub struct ObjectHeader {
    pub group: u8,
    pub variation: u8,
    pub qualifier: QualifierCode,
    pub start_stop_or_count: Vec<u8>,
}

impl ObjectHeader {
    pub fn encode(&self) -> Vec<u8> {
        let mut bytes = vec![self.group, self.variation, self.qualifier as u8];
        bytes.extend_from_slice(&self.start_stop_or_count);
        bytes
    }
}

/// Build a READ request ASDU for a class-based poll.
pub fn build_read_request(sequence: u8, group: u8, variation: u8) -> Vec<u8> {
    let header = AsduHeader {
        function: FunctionCode::Read,
        sequence,
        iin: Iin::default(),
    };
    let mut asdu = header.encode().to_vec();
    // Object header: group, variation, qualifier=Count8, count=1
    asdu.push(group);
    asdu.push(variation);
    asdu.push(QualifierCode::Count8 as u8);
    asdu.push(0x01); // count = 1
    asdu
}

/// Build a READ request for integrity poll (Class 0 = Group 60 Var 1).
pub fn build_integrity_poll_request(sequence: u8) -> Vec<u8> {
    build_read_request(sequence, 60, 1)
}

/// Build a READ request for event scan (Class 1/2/3 = Group 60 Var 2/3/4).
pub fn build_event_scan_request(sequence: u8, class: u8) -> Vec<u8> {
    let variation = match class {
        1 => 2,
        2 => 3,
        3 => 4,
        _ => 1,
    };
    build_read_request(sequence, 60, variation)
}

/// Build a FREEZE request (freeze all counter inputs).
pub fn build_freeze_request(sequence: u8) -> Vec<u8> {
    let header = AsduHeader {
        function: FunctionCode::Freeze,
        sequence,
        iin: Iin::default(),
    };
    let mut asdu = header.encode().to_vec();
    // Object header: Group 20 Var 0 (all counters), qualifier=Count8, count=0
    asdu.push(20);
    asdu.push(0);
    asdu.push(QualifierCode::Count8 as u8);
    asdu.push(0x00);
    asdu
}

/// Build a DIRECT_OPERATE request for binary control output (Group 12 Var 1).
pub fn build_direct_operate_binary(sequence: u8, index: u16, on: bool) -> Vec<u8> {
    let header = AsduHeader {
        function: FunctionCode::DirectOperate,
        sequence,
        iin: Iin::default(),
    };
    let mut asdu = header.encode().to_vec();
    // Object header: Group 12 Var 1, Count=1
    asdu.push(12);
    asdu.push(1);
    asdu.push(QualifierCode::CountAndPrefix8 as u8);
    asdu.push(0x01); // count = 1
    // Prefix: index (1 byte)
    asdu.push(index as u8);
    // Control relay output block: code (1 byte), count (1 byte), on_time (4 bytes), off_time (4 bytes)
    let code = if on { 0x01 } else { 0x02 }; // NUL / NUL_OFF
    asdu.push(code);
    asdu.push(0x01); // count = 1 execution
    asdu.extend_from_slice(&1000u32.to_le_bytes()); // on_time = 1000ms
    asdu.extend_from_slice(&0u32.to_le_bytes()); // off_time = 0
    asdu
}

/// Parse a response ASDU into object headers and data.
pub fn parse_response_asdu(data: &[u8]) -> Result<(AsduHeader, Vec<ParsedObject>)> {
    if data.len() < 4 {
        anyhow::bail!("ASDU too short: {} bytes", data.len());
    }
    let header_bytes: [u8; 4] = [data[0], data[1], data[2], data[3]];
    let header = AsduHeader::decode(&header_bytes)?;

    let mut objects = Vec::new();
    let mut offset = 4;
    while offset + 3 <= data.len() {
        let group = data[offset];
        let variation = data[offset + 1];
        let qualifier_byte = data[offset + 2];
        offset += 3;

        let (count, start_index, prefix_size) = match qualifier_byte {
            0x00 => {
                // Start/Stop 8-bit
                if offset + 2 > data.len() { break; }
                let start = data[offset] as u16;
                let stop = data[offset + 1] as u16;
                offset += 2;
                let count = (stop - start + 1) as usize;
                (count, start, 0)
            }
            0x07 => {
                // Count 8-bit
                if offset + 1 > data.len() { break; }
                let count = data[offset] as usize;
                offset += 1;
                (count, 0, 0)
            }
            0x17 => {
                // Count with prefix 8-bit
                if offset + 1 > data.len() { break; }
                let count = data[offset] as usize;
                offset += 1;
                (count, 0, 1)
            }
            _ => {
                tracing::warn!("Unsupported qualifier: 0x{:02X}", qualifier_byte);
                break;
            }
        };

        let object_data = data[offset..].to_vec();
        objects.push(ParsedObject {
            group,
            variation,
            start_index,
            count,
            prefix_size,
            data: object_data,
        });
        break; // For now, parse first object header only (sufficient for Voltrus polling)
    }

    Ok((header, objects))
}

/// A parsed object from an ASDU response.
#[derive(Debug, Clone)]
pub struct ParsedObject {
    pub group: u8,
    pub variation: u8,
    pub start_index: u16,
    pub count: usize,
    pub prefix_size: usize,
    pub data: Vec<u8>,
}

/// DNP3 master session connected to a single outstation.
pub struct Dnp3MasterSession {
    stream: TcpStream,
    master_addr: u16,
    outstation_addr: u16,
    seq: SequenceTracker,
}

impl Dnp3MasterSession {
    /// Connect to a DNP3 outstation via TCP.
    pub async fn connect(
        ip: &str,
        port: u16,
        master_addr: u16,
        outstation_addr: u16,
    ) -> Result<Self> {
        let addr = format!("{}:{}", ip, port);
        let socket_addr: SocketAddr = addr.parse().context("Invalid DNP3 address")?;
        let stream = tokio::time::timeout(
            std::time::Duration::from_secs(5),
            TcpStream::connect(socket_addr),
        )
        .await
        .context("DNP3 connection timeout")?
        .context("DNP3 connection failed")?;

        let mut session = Self {
            stream,
            master_addr,
            outstation_addr,
            seq: SequenceTracker::new(),
        };

        // Send link reset
        session.send_link_reset().await?;
        // Wait for link reset response
        session.receive_frame().await?;

        Ok(session)
    }

    /// Perform integrity poll (Class 0 — read all static data).
    pub async fn integrity_poll(&mut self) -> Result<HashMap<(u8, u8, u16), Dnp3PointValue>> {
        let seq = self.seq.next();
        let asdu = build_integrity_poll_request(seq);
        self.send_asdu(&asdu).await?;
        self.read_response_objects().await
    }

    /// Perform event scan (Class 1, 2, or 3).
    pub async fn event_scan(&mut self, class: u8) -> Result<HashMap<(u8, u8, u16), Dnp3PointValue>> {
        let seq = self.seq.next();
        let asdu = build_event_scan_request(seq, class);
        self.send_asdu(&asdu).await?;
        self.read_response_objects().await
    }

    /// Freeze all counter inputs.
    pub async fn freeze_counters(&mut self) -> Result<()> {
        let seq = self.seq.next();
        let asdu = build_freeze_request(seq);
        self.send_asdu(&asdu).await?;
        // Read ACK response (may or may not contain data)
        let _ = self.receive_frame().await;
        Ok(())
    }

    /// Send a direct operate command for binary control output.
    pub async fn direct_operate_binary(&mut self, index: u16, on: bool) -> Result<WriteResult> {
        let start = std::time::Instant::now();
        let seq = self.seq.next();
        let asdu = build_direct_operate_binary(seq, index, on);
        self.send_asdu(&asdu).await?;

        match tokio::time::timeout(std::time::Duration::from_secs(5), self.receive_frame()).await {
            Ok(Ok(_)) => Ok(WriteResult {
                success: true,
                device_response: Some(format!("Direct operate: index={} on={}", index, on)),
                error: None,
                latency_ms: start.elapsed().as_millis() as u64,
            }),
            Ok(Err(e)) => Ok(WriteResult {
                success: false,
                device_response: None,
                error: Some(format!("DNP3 direct operate failed: {e}")),
                latency_ms: start.elapsed().as_millis() as u64,
            }),
            Err(_) => Ok(WriteResult {
                success: false,
                device_response: None,
                error: Some("DNP3 direct operate timeout (5s)".into()),
                latency_ms: start.elapsed().as_millis() as u64,
            }),
        }
    }

    async fn send_link_reset(&mut self) -> Result<()> {
        let frame = build_link_reset_frame(self.master_addr, self.outstation_addr);
        let bytes = frame.encode();
        self.stream.write_all(&bytes).await.context("Send link reset failed")?;
        Ok(())
    }

    async fn send_asdu(&mut self, asdu: &[u8]) -> Result<()> {
        let frame = Dnp3Frame {
            master: true,
            fir: true,
            fin: true,
            sequence: 0,
            destination: self.outstation_addr,
            source: self.master_addr,
            data: asdu.to_vec(),
        };
        let bytes = frame.encode();
        self.stream.write_all(&bytes).await.context("Send ASDU failed")?;
        Ok(())
    }

    async fn receive_frame(&mut self) -> Result<Dnp3Frame> {
        let mut header = [0u8; 10]; // min frame: start(2) + length(1) + ctrl(1) + dest(2) + src(2) + crc(2)
        self.stream.read_exact(&mut header[0..3]).await.context("Read frame header failed")?;

        if header[0] != 0x05 || header[1] != 0x64 {
            anyhow::bail!("Invalid DNP3 start bytes");
        }

        let data_len = header[2] as usize;
        let total_frame_size = 3 + data_len + (2 * ((data_len + 15) / 16)); // data + CRC overhead
        let mut buf = vec![0u8; total_frame_size];
        buf[0..3].copy_from_slice(&header[0..3]);
        self.stream.read_exact(&mut buf[3..total_frame_size]).await.context("Read frame body failed")?;

        Dnp3Frame::decode(&buf)
    }

    async fn read_response_objects(&mut self) -> Result<HashMap<(u8, u8, u16), Dnp3PointValue>> {
        let frame = self.receive_frame().await?;
        let (header, parsed_objects) = parse_response_asdu(&frame.data)?;

        if header.function != FunctionCode::Response {
            anyhow::bail!("Expected RESPONSE, got {:?}", header.function);
        }

        let mut results = HashMap::new();
        for obj in parsed_objects {
            let values = objects::parse_objects(obj.group, obj.variation, obj.start_index, obj.count as u16, &obj.data);
            for v in values {
                results.insert((v.group, v.variation, v.index), v);
            }
        }
        Ok(results)
    }
}

use crate::dnp3::master::build_link_reset_frame;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct DiscoveredDnp3Device {
    pub ip: String,
    pub port: u16,
    pub outstation_address: u16,
}

/// Scan a subnet for DNP3 outstations.
/// Sends a link-layer RESET_LINK_STATES to each IP:port.
/// If a response is received, the outstation is discovered.
pub async fn scan_network(cidr: &str, port: u16, timeout_ms: u64) -> Result<Vec<DiscoveredDnp3Device>> {
    let addresses = crate::discovery::parse_cidr(cidr)?;
    let concurrency_limit = 50;
    let timeout = std::time::Duration::from_millis(timeout_ms);
    let semaphore = Arc::new(tokio::sync::Semaphore::new(concurrency_limit));

    let mut handles = Vec::new();
    for ip in addresses {
        let permit = semaphore.clone().acquire_owned().await?;
        let ip_str = ip.to_string();
        handles.push(tokio::spawn(async move {
            let _permit = permit;
            probe_outstation(&ip_str, port, timeout).await
        }));
    }

    let mut results = Vec::new();
    for handle in handles {
        if let Ok(Some(device)) = handle.await {
            results.push(device);
        }
    }

    results.sort_by(|a, b| {
        let a_parts: Vec<u16> = a.ip.split('.').filter_map(|s| s.parse().ok()).collect();
        let b_parts: Vec<u16> = b.ip.split('.').filter_map(|s| s.parse().ok()).collect();
        a_parts.cmp(&b_parts)
    });

    Ok(results)
}

/// Probe a single IP:port for DNP3 outstation.
async fn probe_outstation(ip: &str, port: u16, timeout: std::time::Duration) -> Option<DiscoveredDnp3Device> {
    let addr = format!("{}:{}", ip, port);
    let socket_addr: std::net::SocketAddr = addr.parse().ok()?;

    let connect_result = tokio::time::timeout(timeout, TcpStream::connect(socket_addr)).await;
    let mut stream = match connect_result {
        Ok(Ok(s)) => s,
        _ => return None,
    };

    // Send link reset with master addr=1
    let frame = build_link_reset_frame(1, 0);
    let bytes = frame.encode();
    if stream.write_all(&bytes).await.is_err() {
        return None;
    }

    // Try to read response
    let mut buf = [0u8; 10];
    let read_result = tokio::time::timeout(timeout, stream.read(&mut buf)).await;
    match read_result {
        Ok(Ok(n)) if n >= 8 => {
            // Try to decode as DNP3 frame to extract source address
            if let Ok(response) = Dnp3Frame::decode(&buf[..n]) {
                Some(DiscoveredDnp3Device {
                    ip: ip.to_string(),
                    port,
                    outstation_address: response.source,
                })
            } else {
                // Got bytes but couldn't decode — still consider it a DNP3 device at default addr
                Some(DiscoveredDnp3Device {
                    ip: ip.to_string(),
                    port,
                    outstation_address: 10,
                })
            }
        }
        _ => None,
    }
}

```rust
// src/bin/dnp3-sim.rs

use std::net::SocketAddr;
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::net::TcpListener;

/// DNP3 outstation simulator for testing.
/// Listens on a configurable port and responds to master requests.
#[tokio::main]
async fn main() {
    let port: u16 = std::env::args()
        .nth(1)
        .and_then(|s| s.parse().ok())
        .unwrap_or(20000);
    let outstation_addr: u16 = std::env::args()
        .nth(2)
        .and_then(|s| s.parse().ok())
        .unwrap_or(10);

    let addr = format!("0.0.0.0:{}", port);
    let listener = TcpListener::bind(&addr).await.expect("Failed to bind");
    tracing_subscriber::fmt::init();
    tracing::info!("DNP3 outstation simulator listening on {}, address={}", addr, outstation_addr);

    loop {
        let (mut stream, peer) = listener.accept().await.expect("Accept failed");
        tracing::info!("Connection from {}", peer);

        tokio::spawn(async move {
            let mut buf = [0u8; 4096];
            loop {
                match stream.read(&mut buf).await {
                    Ok(0) => {
                        tracing::info!("Client disconnected");
                        return;
                    }
                    Ok(n) => {
                        tracing::debug!("Received {} bytes", n);
                        // Decode the incoming frame
                        if let Ok(frame) = voltrus_lite_rust_service::dnp3::datalink::Dnp3Frame::decode(&buf[..n]) {
                            tracing::info!(
                                "Frame: master={}, dest={}, src={}, data_len={}",
                                frame.master, frame.destination, frame.source, frame.data.len()
                            );

                            // Build response: swap src/dest, set master=false
                            let response = voltrus_lite_rust_service::dnp3::datalink::Dnp3Frame {
                                master: false,
                                fir: true,
                                fin: true,
                                sequence: frame.sequence,
                                destination: frame.source,
                                source: frame.destination,
                                data: vec![0x81, 0x80, 0x00, 0x00], // RESPONSE with IIN=0
                            };
                            let response_bytes = response.encode();
                            if stream.write_all(&response_bytes).await.is_err() {
                                return;
                            }
                        }
                    }
                    Err(e) => {
                        tracing::error!("Read error: {}", e);
                        return;
                    }
                }
            }
        });
    }
}
```
