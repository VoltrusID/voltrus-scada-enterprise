> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Establish the UI foundation for v0.32.0 -- light/dark theme system, colorblind-friendly status indicators, and multi-language support.

**Architecture:** CSS custom properties with `data-theme` attribute for theming. SVG shape+color StatusIndicator component for accessibility. react-i18next with lazy-loaded JSON for translations. Backend persists user preferences per-user in the `users` table (columns `ui_theme` and `ui_lang`) via the `settings` KV store keyed by `user:{username}:{key}`.

**Tech Stack:** React 19, TypeScript, CSS custom properties, react-i18next, Rust/Axum, SQLite

```rust
    let theme = state
        .storage
        .get_user_preference(&auth.username, "ui_theme")
        .await?
        .unwrap_or_else(|| "dark".to_string());
    Ok(Json(json!({ "theme": theme })))
}

/// PUT /api/v1/settings/ui-theme -- set the authenticated user's theme preference
#[utoipa::path(
    put,
    path = "/api/v1/settings/ui-theme",
    responses((status = 200)),
    tag = "Settings",
    security(("cookie_auth" = []))
)]
pub async fn update_ui_theme(
    State(state): State<AppState>,
    Extension(auth): Extension<AuthUser>,
    Json(body): Json<serde_json::Value>,
) -> Result<impl IntoResponse, AppError> {
    let theme = body
        .get("theme")
        .and_then(|v| v.as_str())
        .unwrap_or("dark");
    let valid = ["light", "dark", "system"].contains(&theme);
    if !valid {
        return Err(AppError(anyhow::anyhow!(
            "Invalid theme. Must be 'light', 'dark', or 'system'."
        )));
    }
    state
        .storage
        .set_user_preference(&auth.username, "ui_theme", theme)
        .await?;

    state.audit_logger.log(crate::security::AuditEvent {
        action: "update_ui_theme".to_string(),
        username: auth.username.clone(),
        resource: "settings:ui-theme".to_string(),
        details: Some(format!("theme={}", theme)),
    });

    Ok(Json(json!({ "ok": true, "theme": theme })))
}
```

```typescript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { queryKeys } from './keys'
import { fetchJson, fetchOk } from './fetch'

export type ThemeValue = 'light' | 'dark' | 'system'

export function useUiTheme() {
  return useQuery({
    queryKey: queryKeys.uiTheme,
    queryFn: () => fetchJson<{ theme: ThemeValue }>('/api/v1/settings/ui-theme'),
    staleTime: Infinity, // theme changes are user-initiated
  })
}

export function useUpdateUiTheme() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: async (theme: ThemeValue) => {
      await fetchOk('/api/v1/settings/ui-theme', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ theme }),
      })
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: queryKeys.uiTheme })
    },
  })
}
```

```typescript
import { createContext, useContext, useEffect, useState, useCallback, type ReactNode } from 'react'
import { useUiTheme, useUpdateUiTheme, type ThemeValue } from './queries'

type ResolvedTheme = 'light' | 'dark'

interface ThemeContextValue {
  theme: ThemeValue          // user preference (light/dark/system)
  resolvedTheme: ResolvedTheme // what is actually applied
  setTheme: (t: ThemeValue) => void
}

const ThemeContext = createContext<ThemeContextValue>({
  theme: 'dark',
  resolvedTheme: 'dark',
  setTheme: () => {},
})

export function useTheme() {
  return useContext(ThemeContext)
}

function getSystemTheme(): ResolvedTheme {
  if (typeof window === 'undefined') return 'dark'
  return window.matchMedia('(prefers-color-scheme: light)').matches ? 'light' : 'dark'
}

function resolveTheme(preference: ThemeValue): ResolvedTheme {
  if (preference === 'system') return getSystemTheme()
  return preference
}

function applyTheme(resolved: ResolvedTheme) {
  document.documentElement.setAttribute('data-theme', resolved)
}

export function ThemeProvider({ children }: { children: ReactNode }) {
  const { data } = useUiTheme()
  const updateThemeMut = useUpdateUiTheme()

  const serverTheme: ThemeValue = data?.theme ?? 'dark'
  const [localTheme, setLocalTheme] = useState<ThemeValue>(() => {
    // Bootstrap from localStorage for instant load before server responds
    const stored = localStorage.getItem('voltrus:ui_theme') as ThemeValue | null
    return stored ?? 'dark'
  })

  // When server responds, sync if different from localStorage
  useEffect(() => {
    if (data) {
      localStorage.setItem('voltrus:ui_theme', data.theme)
      setLocalTheme(data.theme)
    }
  }, [data])

  const resolvedTheme = resolveTheme(localTheme)

  // Apply resolved theme to DOM
  useEffect(() => {
    applyTheme(resolvedTheme)
  }, [resolvedTheme])

  // Listen for system theme changes when in "system" mode
  useEffect(() => {
    if (localTheme !== 'system') return
    const mq = window.matchMedia('(prefers-color-scheme: light)')
    const handler = () => applyTheme(getSystemTheme())
    mq.addEventListener('change', handler)
    return () => mq.removeEventListener('change', handler)
  }, [localTheme])

  const setTheme = useCallback((t: ThemeValue) => {
    setLocalTheme(t)
    localStorage.setItem('voltrus:ui_theme', t)
    updateThemeMut.mutate(t)
  }, [updateThemeMut])

  return (
    <ThemeContext.Provider value={{ theme: localTheme, resolvedTheme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  )
}
```

## Feature 2: Colorblind-Friendly Status Indicators

### Task 2.1 -- StatusIndicator SVG Component

**Files:**
- Create: `client/src/components/StatusIndicator.tsx`

**Steps:**

- [ ] **2.1.1** Create `client/src/components/StatusIndicator.tsx`

  This component renders an SVG indicator that uses both shape and color to convey status, making it accessible to colorblind users.

  The shape mapping follows industrial SCADA conventions:
  - Circle = OK (normal)
  - Triangle = Fault (critical)
  - Diamond = Warning
  - Square = Off (offline/disabled)
  - Question mark circle = Unknown

```typescript
import React from 'react'

export type StatusKind = 'ok' | 'fault' | 'warning' | 'off' | 'unknown'

interface Props {
  status: StatusKind
  size?: 'sm' | 'md' | 'lg'
  label?: string
}

const SIZES: Record<string, number> = { sm: 16, md: 24, lg: 32 }
const STROKES: Record<string, number> = { sm: 1.5, md: 2, lg: 2.5 }

/** Get the fill color from CSS variables based on status */
function statusColor(status: StatusKind): string {
  switch (status) {
    case 'ok': return 'var(--ok)'
    case 'fault': return 'var(--crit)'
    case 'warning': return 'var(--warn)'
    case 'off': return 'var(--muted)'
    case 'unknown': return 'var(--text-dim)'
  }
}

/** Map textual status strings (from backend) to StatusKind */
export function mapStatus(status: string): StatusKind {
  switch (status) {
    case 'ok':
    case 'normal':
    case 'online':
    case 'active':
      return 'ok'
    case 'critical':
    case 'fault':
    case 'error':
      return 'fault'
    case 'warning':
    case 'warn':
      return 'warning'
    case 'offline':
    case 'disabled':
    case 'off':
      return 'off'
    default:
      return 'unknown'
  }
}

/** SVG path for triangle (pointing up, equilateral, centered in viewBox) */
function trianglePath(s: number): string {
  const h = s * 0.866 // sqrt(3)/2
  const cy = s / 2 + s * 0.07 // shift down slightly for visual center
  return `M${s / 2},${cy - h / 2} L${s / 2 + s / 2},${cy + h / 2} L${s / 2 - s / 2},${cy + h / 2} Z`
}

/** SVG path for diamond (rotated square) */
function diamondPath(s: number): string {
  const cx = s / 2, cy = s / 2, r = s / 2 - 1
  return `M${cx},${cy - r} L${cx + r},${cy} L${cx},${cy + r} L${cx - r},${cy} Z`
}

export const StatusIndicator = React.memo(function StatusIndicator({ status, size = 'md', label }: Props) {
  const s = SIZES[size]
  const sw = STROKES[size]
  const color = statusColor(status)
  const strokeColor = 'var(--text)'

  const shape = (() => {
    switch (status) {
      case 'ok':
        // Circle
        return (
          <circle
            cx={s / 2} cy={s / 2} r={s / 2 - 2}
            fill={color} stroke={strokeColor} strokeWidth={sw}
          />
        )
      case 'fault':
        // Triangle
        return (
          <path
            d={trianglePath(s)}
            fill={color} stroke={strokeColor} strokeWidth={sw}
            strokeLinejoin="round"
          />
        )
      case 'warning':
        // Diamond
        return (
          <path
            d={diamondPath(s)}
            fill={color} stroke={strokeColor} strokeWidth={sw}
            strokeLinejoin="round"
          />
        )
      case 'off':
        // Square
        return (
          <rect
            x={2} y={2} width={s - 4} height={s - 4}
            rx={2} fill={color} stroke={strokeColor} strokeWidth={sw}
          />
        )
      case 'unknown':
        // Circle with question mark
        return (
          <g>
            <circle
              cx={s / 2} cy={s / 2} r={s / 2 - 2}
              fill="none" stroke={color} strokeWidth={sw}
            />
            <text
              x={s / 2} y={s / 2}
              textAnchor="middle"
              dominantBaseline="central"
              fill={color}
              fontSize={s * 0.55}
              fontWeight={700}
              fontFamily="Inter, system-ui, sans-serif"
            >
              ?
            </text>
          </g>
        )
    }
  })()

  return (
    <svg
      width={s}
      height={s}
      viewBox={`0 0 ${s} ${s}`}
      role="img"
      aria-label={label ?? `Status: ${status}`}
      title={label ?? status}
    >
      {shape}
    </svg>
  )
})
```

```typescript
import i18n from 'i18next'
import { initReactI18next } from 'react-i18next'
import HttpBackend from 'i18next-http-backend'

const SUPPORTED_LANGS = ['en', 'id', 'zh', 'de'] as const
export type SupportedLang = (typeof SUPPORTED_LANGS)[number]

export { SUPPORTED_LANGS }

// Detect saved language from localStorage, fallback to browser lang, then 'en'
function detectLanguage(): string {
  const stored = localStorage.getItem('voltrus:ui_lang')
  if (stored && SUPPORTED_LANGS.includes(stored as SupportedLang)) return stored

  const browserLang = navigator.language.slice(0, 2)
  if (SUPPORTED_LANGS.includes(browserLang as SupportedLang)) return browserLang

  return 'en'
}

i18n
  .use(HttpBackend)
  .use(initReactI18next)
  .init({
    lng: detectLanguage(),
    fallbackLng: 'en',
    supportedLngs: [...SUPPORTED_LANGS],
    debug: import.meta.env.DEV,
    ns: ['translation'],
    defaultNS: 'translation',
    backend: {
      loadPath: '/locales/{{lng}}/{{ns}}.json',
    },
    interpolation: {
      escapeValue: false, // React already escapes
    },
    react: {
      useSuspense: false, // avoid flash of loading
    },
  })

export default i18n
```

```typescript
import { useTranslation } from 'react-i18next'
import { useUiLang, useUpdateUiLang, type LangValue } from '../queries'

const LANGUAGES: { code: LangValue; label: string }[] = [
  { code: 'en', label: 'English' },
  { code: 'id', label: 'Bahasa Indonesia' },
  { code: 'zh', label: '中文' },
  { code: 'de', label: 'Deutsch' },
]

export function LanguageSelector() {
  const { i18n } = useTranslation()
  const { data } = useUiLang()
  const updateLangMut = useUpdateUiLang()

  const serverLang: LangValue = data?.lang ?? 'en'

  const handleChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    const lang = e.target.value as LangValue
    i18n.changeLanguage(lang)
    localStorage.setItem('voltrus:ui_lang', lang)
    updateLangMut.mutate(lang)
  }

  return (
    <select
      value={serverLang}
      onChange={handleChange}
      aria-label="Language"
      style={{
        background: 'var(--card)',
        border: '1px solid var(--border)',
        color: 'var(--text)',
        borderRadius: 6,
        padding: '6px 10px',
        fontSize: 12,
        fontFamily: 'Inter, system-ui, sans-serif',
        cursor: 'pointer',
        outline: 'none',
        minWidth: 44,
        minHeight: 36,
      }}
    >
      {LANGUAGES.map(({ code, label }) => (
        <option key={code} value={code}>{label}</option>
      ))}
    </select>
  )
}
```
